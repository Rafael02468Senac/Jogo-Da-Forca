358:type GameStatus = 'playing' | 'won' | 'lost'
359:
360:function Game({
361:  onBack,
362:  maxErrors,
363:  onChangeMaxErrors,
364:}: {
365:  onBack: () => void
366:  maxErrors: number
367:  onChangeMaxErrors: (v: number) => void
368:}) {
369:  const [secretWord, setSecretWord]         = useState<string>(() => pickRandom(WORDS_FREE))
370:  const [guessedLetters, setGuessedLetters] = useState<Set<string>>(new Set())
371:  const [gameStatus, setGameStatus]         = useState<GameStatus>('playing')
372:  const [statusMessage, setStatusMessage]   = useState('Adivinhe a palavra — boa sorte!')
373:  const [shakeTrigger, setShakeTrigger]     = useState(0)
374:  const [showSettings, setShowSettings]     = useState(false)
375:
376:  const wrongCount = [...guessedLetters].filter(l => !secretWord.includes(l)).length
377:
378:  // If maxErrors is reduced mid-game and current wrongs exceed new limit → lose
379:  useEffect(() => {
380:    if (gameStatus === 'playing' && wrongCount >= maxErrors) {
381:      setGameStatus('lost')
382:      setStatusMessage(`Você perdeu. A palavra era "${secretWord}".`)
383:    }
384:  }, [maxErrors, wrongCount, gameStatus, secretWord])
385:
386:  const handleGuess = useCallback((raw: string) => {
387:    if (gameStatus !== 'playing') return
388:    const letter = raw.toUpperCase()
389:    if (!/^[A-Z]$/.test(letter)) return
390:    if (guessedLetters.has(letter)) {
391:      setStatusMessage(`A letra "${letter}" já foi utilizada.`)
392:      setShakeTrigger(n => n + 1)
393:      return
394:    }
395:    const next = new Set([...guessedLetters, letter])
396:    setGuessedLetters(next)
397:    if (secretWord.includes(letter)) {
398:      if (secretWord.split('').every(l => next.has(l))) {
399:        setGameStatus('won')
400:        setStatusMessage('Parabéns! Você descobriu a palavra!')
401:      } else {
402:        setStatusMessage(`Boa! "${letter}" está na palavra.`)
403:      }
404:    } else {
405:      const newWrong = [...next].filter(l => !secretWord.includes(l)).length
406:      if (newWrong >= maxErrors) {
407:        setGameStatus('lost')
408:        setStatusMessage(`Você perdeu. A palavra era "${secretWord}".`)
409:      } else {
410:        setStatusMessage(`"${letter}" não está na palavra. +1 erro.`)
411:      }
412:    }
413:  }, [secretWord, guessedLetters, gameStatus, maxErrors])
414:
415:  const newGame = useCallback(() => {
416:    setSecretWord(pickRandom(WORDS_FREE))
417:    setGuessedLetters(new Set())
418:    setGameStatus('playing')
419:    setStatusMessage('Nova partida! Adivinhe a palavra.')
420:  }, [])
421:
422:  useEffect(() => {
